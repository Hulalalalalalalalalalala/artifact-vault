cmake_minimum_required(VERSION 3.20)
project(trustpeek VERSION 0.1.0 LANGUAGES CXX)

find_package(OpenSSL REQUIRED)

add_executable(trustpeek src/main.cpp)
target_compile_features(trustpeek PRIVATE cxx_std_20)
set_target_properties(trustpeek PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(trustpeek PRIVATE OpenSSL::Crypto)

include(CTest)
if(BUILD_TESTING)
    # Only the PEM certificate samples are committed under tests/fixtures/.
    # The regression cases that read DER files and derive the expected
    # fingerprints from those bytes need the DER copies as well; they are
    # prepared OFFLINE in the build tree by base64-decoding each PEM body
    # (tests/prepare_fixtures.sh). No keys are generated and no certificate
    # is re-signed: every prepared <stem>.der is exactly the decoded body of
    # the one committed <stem>.pem. A missing/unreadable PEM or a body that
    # cannot be decoded completely fails the preparation explicitly instead
    # of producing an empty DER or fingerprint.
    set(TRUSTPEEK_FIXTURE_SRC "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    set(TRUSTPEEK_FIXTURE_BINDIR "${CMAKE_CURRENT_BINARY_DIR}/tests")
    set(TRUSTPEEK_FIXTURES_DIR "${TRUSTPEEK_FIXTURE_BINDIR}/fixtures")
    set(TRUSTPEEK_PREPARE "${CMAKE_CURRENT_SOURCE_DIR}/tests/prepare_fixtures.sh")

    # Re-run preparation when the script or any committed PEM sample changes.
    file(GLOB TRUSTPEEK_PEM_FIXTURES CONFIGURE_DEPENDS LIST_DIRECTORIES false
        "${TRUSTPEEK_FIXTURE_SRC}/*.pem")

    # A manifest of the current PEM input set. Make/Ninja do not notice when
    # a prerequisite file is simply DELETED; CONFIGURE_DEPENDS triggers a
    # reconfigure on the glob-set change, rewriting this manifest, which in
    # turn forces preparation to re-run and fail loudly naming the missing
    # sample instead of leaving stale fixtures in the build tree.
    set(TRUSTPEEK_FIXTURE_MANIFEST "${TRUSTPEEK_FIXTURE_BINDIR}/fixtures_manifest.txt")
    set(TRUSTPEEK_FIXTURE_MANIFEST_CONTENT
        "# Generated by CMake; do not edit. Committed PEM fixture inputs:\n")
    foreach(_pem IN LISTS TRUSTPEEK_PEM_FIXTURES)
        string(APPEND TRUSTPEEK_FIXTURE_MANIFEST_CONTENT "${_pem}\n")
    endforeach()
    file(MAKE_DIRECTORY "${TRUSTPEEK_FIXTURE_BINDIR}")
    file(WRITE "${TRUSTPEEK_FIXTURE_MANIFEST}" "${TRUSTPEEK_FIXTURE_MANIFEST_CONTENT}")

    add_custom_command(
        OUTPUT "${TRUSTPEEK_FIXTURES_DIR}/.prepared"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${TRUSTPEEK_FIXTURE_BINDIR}"
        COMMAND sh "${TRUSTPEEK_PREPARE}" "${TRUSTPEEK_FIXTURE_SRC}" "${TRUSTPEEK_FIXTURES_DIR}"
        COMMAND ${CMAKE_COMMAND} -E touch "${TRUSTPEEK_FIXTURES_DIR}/.prepared"
        DEPENDS "${TRUSTPEEK_PREPARE}" "${TRUSTPEEK_FIXTURE_MANIFEST}"
                ${TRUSTPEEK_PEM_FIXTURES}
        COMMENT "Preparing regression-test certificate fixtures (PEM -> DER)"
        VERBATIM
    )
    # Part of the normal build: after `cmake --build`, the offline build tree
    # already contains every certificate byte the regression checks need.
    add_custom_target(trustpeek-fixtures ALL
        DEPENDS "${TRUSTPEEK_FIXTURES_DIR}/.prepared")

    set(TRUSTPEEK_CASES

    # Defense in depth for test runs against an already-configured build
    # directory: the setup test prepares the fixtures (and fails loudly if a
    # sample is unusable) before any inspect case may start. Every registered
    # case keeps its original command and comparisons; only the fixture
    # directory it reads becomes the prepared one in the build tree.
    add_test(
        NAME prepare_fixtures
        COMMAND sh "${TRUSTPEEK_PREPARE}" "${TRUSTPEEK_FIXTURE_SRC}" "${TRUSTPEEK_FIXTURES_DIR}"
    )
    set_tests_properties(prepare_fixtures PROPERTIES
        FIXTURES_SETUP TRUSTPEEK_FIXTURES)

    foreach(case IN LISTS TRUSTPEEK_CASES)
        add_test(
            NAME inspect_${case}
            COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_case.sh
                    $<TARGET_FILE:trustpeek>
                    ${TRUSTPEEK_FIXTURES_DIR}
                    ${case}
        )
        set_tests_properties(inspect_${case} PROPERTIES
            FIXTURES_REQUIRED TRUSTPEEK_FIXTURES)
    endforeach()
